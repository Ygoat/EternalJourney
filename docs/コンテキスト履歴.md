         [Export] public int InitialSize { get; set; } = 50;
         [Export] public int MaxSize { get; set; } = 500;
         
         private readonly Queue<T> _available = new();
         private readonly HashSet<T> _inUse = new();
         
         public T? Acquire()
         {
             T? instance;
             if (_available.Count > 0)
                 instance = _available.Dequeue();
             else if (AutoGrow && TotalCount < MaxSize)
                 instance = CreateInstance();
             else
                 return null;

             return instance;
